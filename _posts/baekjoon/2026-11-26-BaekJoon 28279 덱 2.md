## 1. 아이디어

[[BaekJoon] #10866 - 덱](/posts/baekjoon-10866/)과 같은 정수 덱에서 명령이 숫자로 바뀌고 명령의 수 $N$이 1,000,000까지 늘어난 문제로, 배열의 첫 칸을 항상 맨 앞으로 두면 앞에서 넣고 뺄 때마다 원소를 밀어야 해 최악의 경우 $O(N^2)$이 된다. 따라서 원소는 그대로 두고 맨 앞 원소의 인덱스와 맨 뒤 원소의 다음 인덱스만 옮긴다. 이때 원소가 놓인 구간은 앞에 넣으면 왼쪽으로, 뒤에 넣으면 오른쪽으로 늘어나는데 어느 쪽도 $N$칸을 넘지 않으므로, 크기 2,000,000인 배열의 가운데에서 빈 상태로 시작하면 범위를 벗어나지 않고 모든 연산을 상수 시간에 처리할 수 있다.

## 2. 코드

### 풀이 [Java][C++][Python]

```java
import java.io.*;
import java.util.*;

public class Main {

    static final int MAX_N = 1_000_000;
    static int[] deque = new int[MAX_N * 2];
    static int front = MAX_N, rear = MAX_N;

    static void push_front(int x) {
        deque[--front] = x;
    }

    static void push_back(int x) {
        deque[rear++] = x;
    }

    static int pop_front() {
        return deque[front++];
    }

    static int pop_back() {
        return deque[--rear];
    }

    static int size() {
        return rear - front;
    }

    static boolean empty() {
        return front == rear;
    }

    static int front() {
        return deque[front];
    }

    static int back() {
        return deque[rear - 1];
    }

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        StringBuilder sb = new StringBuilder();
        StringTokenizer st;

        int n = Integer.parseInt(br.readLine());
        while (n-- > 0) {
            st = new StringTokenizer(br.readLine());
            int op = Integer.parseInt(st.nextToken());

            if (op == 1) {
                int x = Integer.parseInt(st.nextToken());
                push_front(x);
            } else if (op == 2) {
                int x = Integer.parseInt(st.nextToken());
                push_back(x);
            } else if (op == 3) {
                if (empty()) {
                    sb.append("-1\n");
                } else {
                    sb.append(pop_front()).append("\n");
                }
            } else if (op == 4) {
                if (empty()) {
                    sb.append("-1\n");
                } else {
                    sb.append(pop_back()).append("\n");
                }
            } else if (op == 5) {
                sb.append(size()).append("\n");
            } else if (op == 6) {
                if (empty()) {
                    sb.append("1\n");
                } else {
                    sb.append("0\n");
                }
            } else if (op == 7) {
                if (empty()) {
                    sb.append("-1\n");
                } else {
                    sb.append(front()).append("\n");
                }
            } else {
                if (empty()) {
                    sb.append("-1\n");
                } else {
                    sb.append(back()).append("\n");
                }
            }
        }

        System.out.println(sb);
    }
}
```

`bool`은 `cout`에서 1과 0으로 출력되므로 `empty`의 반환값을 분기 없이 그대로 썼다.

```python
import sys

input = sys.stdin.readline
out = []

MAX_N = 1_000_000


class Deque:
    def __init__(self):
        self.dq = [0] * (MAX_N * 2)
        self.s = MAX_N
        self.e = MAX_N

    def push_front(self, x):
        self.s -= 1
        self.dq[self.s] = x

    def push_back(self, x):
        self.dq[self.e] = x
        self.e += 1

    def pop_front(self):
        self.s += 1
        return self.dq[self.s - 1]

    def pop_back(self):
        self.e -= 1
        return self.dq[self.e]

    def size(self):
        return self.e - self.s

    def empty(self):
        return self.s == self.e

    def front(self):
        return self.dq[self.s]

    def back(self):
        return self.dq[self.e - 1]


deque = Deque()

n = int(input())
for _ in range(n):
    tokens = input().split()
    op = tokens[0]

    if op == "1":
        deque.push_front(int(tokens[1]))
    elif op == "2":
        deque.push_back(int(tokens[1]))
    elif op == "3":
        if deque.empty():
            out.append(-1)
        else:
            out.append(deque.pop_front())
    elif op == "4":
        if deque.empty():
            out.append(-1)
        else:
            out.append(deque.pop_back())
    elif op == "5":
        out.append(deque.size())
    elif op == "6":
        if deque.empty():
            out.append(1)
        else:
            out.append(0)
    elif op == "7":
        if deque.empty():
            out.append(-1)
        else:
            out.append(deque.front())
    else:
        if deque.empty():
            out.append(-1)
        else:
            out.append(deque.back())

print("\n".join(map(str, out)))
```
