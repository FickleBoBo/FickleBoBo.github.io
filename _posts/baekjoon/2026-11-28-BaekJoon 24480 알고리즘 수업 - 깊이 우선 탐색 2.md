## 1. 아이디어

[[BaekJoon] #24479 - 알고리즘 수업 - 깊이 우선 탐색 1](/posts/baekjoon-24479/)과 같이 정점 $R$에서 DFS를 했을 때 각 정점이 몇 번째로 방문되는지 구하는 문제인데, 인접 정점을 방문하는 순서만 번호 내림차순으로 바뀌었다. DFS는 인접 리스트에 놓인 순서대로 다음 정점을 고르므로, 각 정점의 인접 리스트를 내림차순으로 정렬해 두고 DFS를 하며 정점에 들어설 때마다 1씩 늘어나는 번호를 매기면 방문 순서가 된다.

## 2. 코드

### 풀이 [Java][C++][Python]

```java
import java.io.*;
import java.util.*;

public class Main {

    static List<Integer>[] adj;
    static boolean[] vis;
    static int[] order;
    static int cnt = 1;

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        StringBuilder sb = new StringBuilder();
        StringTokenizer st = new StringTokenizer(br.readLine());

        int n = Integer.parseInt(st.nextToken());
        int m = Integer.parseInt(st.nextToken());
        int r = Integer.parseInt(st.nextToken());

        adj = new ArrayList[1 + n];
        for (int i = 1; i <= n; i++) {
            adj[i] = new ArrayList<>();
        }

        while (m-- > 0) {
            st = new StringTokenizer(br.readLine());
            int u = Integer.parseInt(st.nextToken());
            int v = Integer.parseInt(st.nextToken());
            adj[u].add(v);
            adj[v].add(u);
        }

        for (int i = 1; i <= n; i++) {
            adj[i].sort(Comparator.reverseOrder());
        }

        vis = new boolean[1 + n];
        order = new int[1 + n];

        dfs(r);

        for (int i = 1; i <= n; i++) {
            sb.append(order[i]).append("\n");
        }

        System.out.println(sb);
    }

    static void dfs(int cur) {
        vis[cur] = true;
        order[cur] = cnt++;

        for (int nxt : adj[cur]) {
            if (vis[nxt]) continue;
            dfs(nxt);
        }
    }
}
```

`Comparator.reverseOrder()`를 `sort`에 넘겨 내림차순으로 정렬했다.

```c++
#include <bits/stdc++.h>
using namespace std;

const int MAX_N = 1 + 100000;
vector<int> adj[MAX_N];
bool vis[MAX_N];
int order[MAX_N];
int cnt = 1;

void dfs(int cur) {
    vis[cur] = true;
    order[cur] = cnt++;

    for (int nxt : adj[cur]) {
        if (vis[nxt]) continue;
        dfs(nxt);
    }
}

int main() {
    ios::sync_with_stdio(0);
    cin.tie(0);

    int n, m, r;
    cin >> n >> m >> r;

    while (m--) {
        int u, v;
        cin >> u >> v;
        adj[u].push_back(v);
        adj[v].push_back(u);
    }

    for (int i = 1; i <= n; i++) {
        sort(adj[i].rbegin(), adj[i].rend());
    }

    dfs(r);

    for (int i = 1; i <= n; i++) {
        cout << order[i] << '\n';
    }
}
```

`std::sort`에 역방향 이터레이터 `rbegin`, `rend`를 넘겨 별도 비교 함수 없이 내림차순으로 정렬했다.

```python
import sys

input = sys.stdin.readline

sys.setrecursionlimit(10**6)


def dfs(cur):
    global cnt
    vis[cur] = True
    order[cur] = cnt
    cnt += 1

    for nxt in adj[cur]:
        if not vis[nxt]:
            dfs(nxt)


n, m, r = map(int, input().split())
adj = [[] for _ in range(1 + n)]
for _ in range(m):
    u, v = map(int, input().split())
    adj[u].append(v)
    adj[v].append(u)

for x in adj:
    x.sort(reverse=True)

vis = [False] * (1 + n)
order = [0] * (1 + n)
cnt = 1

dfs(r)

print("\n".join(map(str, order[1:])))
```

정점이 최대 100,000개라 재귀가 그만큼 깊어질 수 있어 `sys.setrecursionlimit`으로 기본 한도를 올렸다.
