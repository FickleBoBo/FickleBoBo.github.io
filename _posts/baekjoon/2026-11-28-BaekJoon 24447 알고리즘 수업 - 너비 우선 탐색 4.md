## 1. 아이디어

[[BaekJoon] #24446 - 알고리즘 수업 - 너비 우선 탐색 3](/posts/baekjoon-24446/)처럼 정점 $R$에서 BFS를 하되 인접 정점을 번호 오름차순으로 방문하며, 각 정점의 깊이 $d_i$와 방문 순서 $t_i$를 곱한 값을 모든 정점에 대해 더하는 문제다. BFS 한 번에서 새 정점을 발견할 때 깊이를, 큐에서 꺼낼 때 순서를 기록하면 두 값이 함께 구해진다. 이때 깊이와 달리 방문 순서는 인접 정점을 살피는 순서에 따라 달라지므로 인접 리스트를 오름차순으로 정렬해야 한다. 닿지 못한 정점은 $d_i = -1$, $t_i = 0$이라 곱이 0이 되어 따로 걸러내지 않아도 된다. 깊이는 방문 순서에서 1을 뺀 값을 넘지 못하는데, 정점들이 $R$을 한쪽 끝으로 일렬로 이어지면 모든 정점이 이 상한에 닿아 값이 가장 커진다. $N$이 최대 100,000이라 이때 곱 하나가 약 $10^{10}$, 합이 약 $3.3 \times 10^{14}$까지 커지므로 64비트 정수로 계산한다.

## 2. 코드

### 풀이 [Java][C++][Python]

```java
import java.io.*;
import java.util.*;

public class Main {

    static int n;
    static List<Integer>[] adj;

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        StringTokenizer st = new StringTokenizer(br.readLine());

        n = Integer.parseInt(st.nextToken());
        int m = Integer.parseInt(st.nextToken());
        int r = Integer.parseInt(st.nextToken());

        adj = new ArrayList[1 + n];
        for (int i = 1; i <= n; i++) {
            adj[i] = new ArrayList<>();
        }

        while (m-- > 0) {
            st = new StringTokenizer(br.readLine());
            int u = Integer.parseInt(st.nextToken());
            int v = Integer.parseInt(st.nextToken());
            adj[u].add(v);
            adj[v].add(u);
        }

        for (int i = 1; i <= n; i++) {
            adj[i].sort(Comparator.naturalOrder());
        }

        System.out.println(bfs(r));
    }

    static long bfs(int start) {
        Queue<Integer> q = new ArrayDeque<>();
        q.offer(start);

        int[] dist = new int[1 + n];
        Arrays.fill(dist, -1);
        dist[start] = 0;

        int[] order = new int[1 + n];
        int cnt = 1;

        while (!q.isEmpty()) {
            int cur = q.poll();
            order[cur] = cnt++;

            for (int nxt : adj[cur]) {
                if (dist[nxt] != -1) continue;
                q.offer(nxt);
                dist[nxt] = dist[cur] + 1;
            }
        }

        long ans = 0;
        for (int i = 1; i <= n; i++) {
            ans += (long) dist[i] * order[i];
        }

        return ans;
    }
}
```

`dist[i] * order[i]`가 `int` 범위를 넘을 수 있어 `dist[i]`를 `long`으로 형변환한 뒤 곱했다.

```c++
#include <bits/stdc++.h>
using namespace std;

const int MAX_N = 1 + 100000;
vector<int> adj[MAX_N];
int dist[MAX_N];
int order[MAX_N];

void bfs(int start) {
    queue<int> q;
    q.push(start);

    memset(dist, -1, sizeof(dist));
    dist[start] = 0;

    int cnt = 1;

    while (!q.empty()) {
        int cur = q.front();
        q.pop();

        order[cur] = cnt++;

        for (int nxt : adj[cur]) {
            if (dist[nxt] != -1) continue;
            q.push(nxt);
            dist[nxt] = dist[cur] + 1;
        }
    }
}

int main() {
    ios::sync_with_stdio(0);
    cin.tie(0);

    int n, m, r;
    cin >> n >> m >> r;

    while (m--) {
        int u, v;
        cin >> u >> v;
        adj[u].push_back(v);
        adj[v].push_back(u);
    }

    for (int i = 1; i <= n; i++) {
        sort(adj[i].begin(), adj[i].end());
    }

    bfs(r);

    long long ans = 0;
    for (int i = 1; i <= n; i++) {
        ans += 1LL * dist[i] * order[i];
    }

    cout << ans;
}
```

`std::memset`은 바이트 단위로 채우는데, `-1`은 모든 비트가 1이라 `int` 배열도 `-1`로 채워졌다. `dist[i] * order[i]`가 `int` 범위를 넘을 수 있어 `1LL`을 먼저 곱해 `long long`으로 계산했다.

```python
import sys
from collections import deque

input = sys.stdin.readline


def bfs(start):
    q = deque([start])

    dist = [-1] * (1 + n)
    dist[start] = 0

    order = [0] * (1 + n)
    cnt = 1

    while q:
        cur = q.popleft()
        order[cur] = cnt
        cnt += 1

        for nxt in adj[cur]:
            if dist[nxt] != -1:
                continue
            dist[nxt] = dist[cur] + 1
            q.append(nxt)

    return sum(dist[i] * order[i] for i in range(1, n + 1))


n, m, r = map(int, input().split())
adj = [[] for _ in range(1 + n)]
for _ in range(m):
    u, v = map(int, input().split())
    adj[u].append(v)
    adj[v].append(u)

for x in adj:
    x.sort()

print(bfs(r))
```
