## 1. 아이디어

정수 덱을 구현해 명령을 처리하는 문제로, 덱은 양쪽 끝에서 넣고 뺄 수 있어 배열로 구현하려면 앞쪽에도 늘어날 자리가 있어야 한다. 명령이 10,000개 이하라 어느 쪽으로든 늘어나는 칸 수가 10,000을 넘지 않으므로, 크기 20,000인 배열의 가운데에서 빈 상태로 시작하면 범위를 벗어나지 않는다. 맨 앞 원소의 인덱스와 맨 뒤 원소의 다음 인덱스를 관리하면 어느 쪽에서 넣고 빼든 인덱스 하나만 움직이고, 두 인덱스의 차이가 덱의 크기이다.

## 2. 코드

### 풀이 [Java][C++][Python]

```java
import java.io.*;
import java.util.*;

public class Main {

    static final int MAX_N = 10000;
    static int[] deque = new int[MAX_N * 2];
    static int front = MAX_N, rear = MAX_N;

    static void push_front(int x) {
        deque[--front] = x;
    }

    static void push_back(int x) {
        deque[rear++] = x;
    }

    static int pop_front() {
        return deque[front++];
    }

    static int pop_back() {
        return deque[--rear];
    }

    static int size() {
        return rear - front;
    }

    static boolean empty() {
        return front == rear;
    }

    static int front() {
        return deque[front];
    }

    static int back() {
        return deque[rear - 1];
    }

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        StringBuilder sb = new StringBuilder();
        StringTokenizer st;

        int n = Integer.parseInt(br.readLine());
        while (n-- > 0) {
            st = new StringTokenizer(br.readLine());
            String q = st.nextToken();

            if (q.equals("push_front")) {
                int x = Integer.parseInt(st.nextToken());
                push_front(x);
            } else if (q.equals("push_back")) {
                int x = Integer.parseInt(st.nextToken());
                push_back(x);
            } else if (q.equals("pop_front")) {
                if (empty()) {
                    sb.append("-1\n");
                } else {
                    sb.append(pop_front()).append("\n");
                }
            } else if (q.equals("pop_back")) {
                if (empty()) {
                    sb.append("-1\n");
                } else {
                    sb.append(pop_back()).append("\n");
                }
            } else if (q.equals("size")) {
                sb.append(size()).append("\n");
            } else if (q.equals("empty")) {
                if (empty()) {
                    sb.append("1\n");
                } else {
                    sb.append("0\n");
                }
            } else if (q.equals("front")) {
                if (empty()) {
                    sb.append("-1\n");
                } else {
                    sb.append(front()).append("\n");
                }
            } else {
                if (empty()) {
                    sb.append("-1\n");
                } else {
                    sb.append(back()).append("\n");
                }
            }
        }

        System.out.println(sb);
    }
}
```

`bool`은 `cout`에서 1과 0으로 출력되므로 `empty`의 반환값을 분기 없이 그대로 썼다.

```python
import sys

input = sys.stdin.readline

MAX_N = 10000


class Deque:
    def __init__(self):
        self.dq = [0] * (MAX_N * 2)
        self.s = MAX_N
        self.e = MAX_N

    def push_front(self, x):
        self.s -= 1
        self.dq[self.s] = x

    def push_back(self, x):
        self.dq[self.e] = x
        self.e += 1

    def pop_front(self):
        self.s += 1
        return self.dq[self.s - 1]

    def pop_back(self):
        self.e -= 1
        return self.dq[self.e]

    def size(self):
        return self.e - self.s

    def empty(self):
        return self.s == self.e

    def front(self):
        return self.dq[self.s]

    def back(self):
        return self.dq[self.e - 1]


deque = Deque()
out = []

n = int(input())
for _ in range(n):
    q = input().split()

    if q[0] == "push_front":
        deque.push_front(int(q[1]))
    elif q[0] == "push_back":
        deque.push_back(int(q[1]))
    elif q[0] == "pop_front":
        if deque.empty():
            out.append(-1)
        else:
            out.append(deque.pop_front())
    elif q[0] == "pop_back":
        if deque.empty():
            out.append(-1)
        else:
            out.append(deque.pop_back())
    elif q[0] == "size":
        out.append(deque.size())
    elif q[0] == "empty":
        if deque.empty():
            out.append(1)
        else:
            out.append(0)
    elif q[0] == "front":
        if deque.empty():
            out.append(-1)
        else:
            out.append(deque.front())
    else:
        if deque.empty():
            out.append(-1)
        else:
            out.append(deque.back())

print("\n".join(map(str, out)))
```
