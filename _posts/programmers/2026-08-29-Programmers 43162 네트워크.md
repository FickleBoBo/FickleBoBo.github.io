## 1. 아이디어

컴퓨터 A, B가 연결되고 B, C가 연결되면 A, C도 같은 네트워크이므로 네트워크는 서로 닿을 수 있는 컴퓨터끼리의 묶음이고, 결국 인접 행렬로 주어진 그래프의 연결 요소 개수를 세는 문제다.

BFS는 아직 방문하지 않은 컴퓨터에서 시작하면 닿는 컴퓨터가 모두 방문 처리되어 한 네트워크가 통째로 소진되므로 BFS를 시작한 횟수가 곧 네트워크 수이다. DFS는 같은 탐색을 깊이 우선으로 바꾼 것으로 $n \le 200$이라 재귀 깊이가 문제 되지 않아 BFS와 동등한 대안이다.

유니온 파인드는 연결된 두 컴퓨터를 합쳐 가면 같은 네트워크의 컴퓨터가 하나의 대표를 공유하게 되므로 대표가 자기 자신인 컴퓨터의 수가 네트워크 수이다. 연결은 양방향이라 인접 행렬이 대칭이므로 위쪽 삼각형만 훑어 합치면 되고, 탐색 대신 합집합 연산으로 푸는 것일 뿐 인접 행렬을 훑는 비용은 같다.

## 2. 코드

### 풀이 1: BFS [Java][C++][Python]

```java
import java.util.*;

class Solution {
    public int solution(int n, int[][] computers) {
        boolean[] vis = new boolean[n];
        int cnt = 0;

        for (int i = 0; i < n; i++) {
            if (vis[i]) continue;
            bfs(i, n, computers, vis);
            cnt++;
        }

        return cnt;
    }

    static void bfs(int start, int n, int[][] computers, boolean[] vis) {
        Queue<Integer> q = new ArrayDeque<>();
        q.offer(start);

        vis[start] = true;

        while (!q.isEmpty()) {
            int cur = q.poll();

            for (int nxt = 0; nxt < n; nxt++) {
                if (computers[cur][nxt] == 0 || vis[nxt]) continue;
                q.offer(nxt);
                vis[nxt] = true;
            }
        }
    }
}
```

```c++
#include <bits/stdc++.h>
using namespace std;

const int MAX_N = 200;
bool vis[MAX_N];

void bfs(int start, int n, vector<vector<int>>& computers) {
    queue<int> q;
    q.push(start);

    vis[start] = true;

    while (!q.empty()) {
        int cur = q.front();
        q.pop();

        for (int nxt = 0; nxt < n; nxt++) {
            if (computers[cur][nxt] == 0 || vis[nxt]) continue;
            q.push(nxt);
            vis[nxt] = true;
        }
    }
}

int solution(int n, vector<vector<int>> computers) {
    int cnt = 0;

    for (int i = 0; i < n; i++) {
        if (vis[i]) continue;
        bfs(i, n, computers);
        cnt++;
    }

    return cnt;
}
```

```python
from collections import deque


def solution(n, computers):
    vis = [False] * n

    def bfs(start):
        q = deque([start])
        vis[start] = True

        while q:
            cur = q.popleft()

            for nxt in range(n):
                if computers[cur][nxt] == 0 or vis[nxt]:
                    continue
                q.append(nxt)
                vis[nxt] = True

    cnt = 0

    for i in range(n):
        if vis[i]:
            continue
        bfs(i)
        cnt += 1

    return cnt
```

### 풀이 3: 유니온 파인드 [Java][C++][Python]

```java
class Solution {

    static int[] p;

    static int find(int x) {
        if (x == p[x]) return x;
        return p[x] = find(p[x]);
    }

    static boolean union(int x, int y) {
        x = find(x);
        y = find(y);
        if (x == y) return false;
        p[x] = y;
        return true;
    }

    public int solution(int n, int[][] computers) {
        p = new int[n];
        for (int i = 0; i < n; i++) {
            p[i] = i;
        }

        for (int i = 0; i < n - 1; i++) {
            for (int j = i + 1; j < n; j++) {
                if (computers[i][j] == 1) union(i, j);
            }
        }

        int cnt = 0;
        for (int i = 0; i < n; i++) {
            if (i == p[i]) cnt++;
        }

        return cnt;
    }
}
```

```c++
#include <bits/stdc++.h>
using namespace std;

const int MAX_N = 200;
vector<int> p(MAX_N);

int find(int x) {
    if (x == p[x]) return x;
    return p[x] = find(p[x]);
}

bool unite(int x, int y) {
    x = find(x);
    y = find(y);
    if (x == y) return false;
    p[x] = y;
    return true;
}

int solution(int n, vector<vector<int>> computers) {
    iota(p.begin(), p.end(), 0);

    for (int i = 0; i < n - 1; i++) {
        for (int j = i + 1; j < n; j++) {
            if (computers[i][j]) unite(i, j);
        }
    }

    int cnt = 0;
    for (int i = 0; i < n; i++) {
        if (i == p[i]) cnt++;
    }

    return cnt;
}
```

`union`은 C++ 예약어라 함수 이름을 `unite`로 지었다.

```python
def solution(n, computers):
    p = list(range(n))

    def find(x):
        while x != p[x]:
            p[x] = p[p[x]]
            x = p[x]
        return x

    def union(x, y):
        x, y = find(x), find(y)
        if x == y:
            return False
        p[x] = y
        return True

    for i in range(n - 1):
        for j in range(i + 1, n):
            if computers[i][j]:
                union(i, j)

    return sum(i == p[i] for i in range(n))
```
