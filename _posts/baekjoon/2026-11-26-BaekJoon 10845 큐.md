## 1. 아이디어

정수 큐를 구현해 명령을 처리하는 문제로, 큐는 뒤에 넣고 앞에서 꺼내므로 배열에 값을 차례로 쓰면서 맨 앞 원소의 인덱스와 다음에 넣을 인덱스를 따로 관리한다. 꺼낼 때는 남은 원소를 당기지 않고 앞 인덱스만 한 칸 뒤로 옮기면 되고, 두 인덱스의 차이가 큐의 크기이며 차이가 0이면 빈 큐이다. 명령이 10,000개 이하라 넣는 횟수도 10,000번을 넘지 않으므로, 꺼내서 비게 된 앞쪽 칸을 원형으로 다시 쓰지 않아도 크기 10,000인 배열이면 충분하다.

## 2. 코드

### 풀이 [Java][C++][Python]

```java
import java.io.*;
import java.util.*;

public class Main {

    static final int MAX_N = 10000;
    static int[] queue = new int[MAX_N];
    static int front = 0, rear = 0;

    static void push(int x) {
        queue[rear++] = x;
    }

    static int pop() {
        return queue[front++];
    }

    static int size() {
        return rear - front;
    }

    static boolean empty() {
        return front == rear;
    }

    static int front() {
        return queue[front];
    }

    static int back() {
        return queue[rear - 1];
    }

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        StringBuilder sb = new StringBuilder();
        StringTokenizer st;

        int n = Integer.parseInt(br.readLine());
        while (n-- > 0) {
            st = new StringTokenizer(br.readLine());
            String q = st.nextToken();

            if (q.equals("push")) {
                int x = Integer.parseInt(st.nextToken());
                push(x);
            } else if (q.equals("pop")) {
                if (empty()) {
                    sb.append("-1\n");
                } else {
                    sb.append(pop()).append("\n");
                }
            } else if (q.equals("size")) {
                sb.append(size()).append("\n");
            } else if (q.equals("empty")) {
                if (empty()) {
                    sb.append("1\n");
                } else {
                    sb.append("0\n");
                }
            } else if (q.equals("front")) {
                if (empty()) {
                    sb.append("-1\n");
                } else {
                    sb.append(front()).append("\n");
                }
            } else {
                if (empty()) {
                    sb.append("-1\n");
                } else {
                    sb.append(back()).append("\n");
                }
            }
        }

        System.out.println(sb);
    }
}
```

`bool`은 `cout`에서 1과 0으로 출력되므로 `empty`의 반환값을 분기 없이 그대로 썼다.

```python
import sys

input = sys.stdin.readline

MAX_N = 10000


class Queue:
    def __init__(self):
        self.q = [0] * MAX_N
        self.s = 0
        self.e = 0

    def push(self, x):
        self.q[self.e] = x
        self.e += 1

    def pop(self):
        self.s += 1
        return self.q[self.s - 1]

    def size(self):
        return self.e - self.s

    def empty(self):
        return self.s == self.e

    def front(self):
        return self.q[self.s]

    def back(self):
        return self.q[self.e - 1]


queue = Queue()
out = []

n = int(input())
for _ in range(n):
    q = input().split()

    if q[0] == "push":
        queue.push(int(q[1]))
    elif q[0] == "pop":
        if queue.empty():
            out.append(-1)
        else:
            out.append(queue.pop())
    elif q[0] == "size":
        out.append(queue.size())
    elif q[0] == "empty":
        if queue.empty():
            out.append(1)
        else:
            out.append(0)
    elif q[0] == "front":
        if queue.empty():
            out.append(-1)
        else:
            out.append(queue.front())
    else:
        if queue.empty():
            out.append(-1)
        else:
            out.append(queue.back())

print("\n".join(map(str, out)))
```
