## 1. 아이디어

각 지점에서 목표지점까지의 최단 거리를 구하는 문제로 BFS를 이용하면 해결할 수 있다. 각 지점에서 매번 목표 지점까지의 거리를 구하는 것이 아니라 목표 지점에서 각 지점까지의 거리를 구하는 방식으로 한 번의 BFS로 모든 지점까지의 거리를 구할 수 있으며 이를 2차원 배열 `dist`에 저장했다. `dist`의 경우 초기 갈 수 있는 땅은 -1로 초기화하고 목표지점은 0으로 초기화한 상태에서 BFS 탐색 중 다음 노드의 거리는 이전 노드의 거리에 1을 더하는 방식으로 하면 되는데, 마침 갈 수 있는 땅이 1로 주어져 있어서 부호만 뒤집어서 대입하는 방식으로 초기화했다.

---

## 2. 복잡도

| 접근 | 시간            | 공간            |
| ---- | --------------- | --------------- |
| 풀이 | $O(N \times M)$ | $O(N \times M)$ |

($N$ = 입력값 `n`, $M$ = 입력값 `m`)

---

## 3. 코드

### 풀이 [Java][C++]

```java
import java.io.*;
import java.util.*;

public class Main {

    static int[] dr = {-1, 0, 1, 0};
    static int[] dc = {0, 1, 0, -1};
    static int n, m;
    static int[][] grid;

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        StringBuilder sb = new StringBuilder();
        StringTokenizer st = new StringTokenizer(br.readLine());

        n = Integer.parseInt(st.nextToken());
        m = Integer.parseInt(st.nextToken());

        grid = new int[n][m];
        int sr = -1;
        int sc = -1;

        for (int i = 0; i < n; i++) {
            st = new StringTokenizer(br.readLine());
            for (int j = 0; j < m; j++) {
                grid[i][j] = Integer.parseInt(st.nextToken());
                if (grid[i][j] == 2) {
                    sr = i;
                    sc = j;
                }
            }
        }

        int[][] dist = bfs(sr, sc);
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                sb.append(dist[i][j]).append(" ");
            }
            sb.append("\n");
        }

        System.out.println(sb);
    }

    static int[][] bfs(int sr, int sc) {
        Queue<int[]> q = new ArrayDeque<>();
        q.offer(new int[]{sr, sc});

        int[][] dist = new int[n][m];
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                dist[i][j] = -grid[i][j];
            }
        }
        dist[sr][sc] = 0;

        while (!q.isEmpty()) {
            int[] cur = q.poll();

            for (int d = 0; d < 4; d++) {
                int nr = cur[0] + dr[d];
                int nc = cur[1] + dc[d];

                if (nr < 0 || nr >= n || nc < 0 || nc >= m) continue;
                if (grid[nr][nc] == 0 || dist[nr][nc] != -1) continue;

                q.offer(new int[]{nr, nc});
                dist[nr][nc] = dist[cur[0]][cur[1]] + 1;
            }
        }

        return dist;
    }
}
```

```c++
#include <bits/stdc++.h>
using namespace std;

int dr[4] = {-1, 0, 1, 0};
int dc[4] = {0, 1, 0, -1};
int n, m;
int grid[1000][1000];
int dist[1000][1000];

void bfs(int sr, int sc) {
    queue<pair<int, int>> q;
    q.push({sr, sc});

    for (int i = 0; i < n; i++) {
        for (int j = 0; j < m; j++) {
            dist[i][j] = -grid[i][j];
        }
    }
    dist[sr][sc] = 0;

    while (!q.empty()) {
        auto [r, c] = q.front();
        q.pop();

        for (int d = 0; d < 4; d++) {
            int nr = r + dr[d];
            int nc = c + dc[d];

            if (nr < 0 || nr >= n || nc < 0 || nc >= m) continue;
            if (grid[nr][nc] == 0 || dist[nr][nc] != -1) continue;

            q.push({nr, nc});
            dist[nr][nc] = dist[r][c] + 1;
        }
    }
}

int main() {
    ios::sync_with_stdio(0);
    cin.tie(0);

    cin >> n >> m;

    int sr = -1;
    int sc = -1;
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < m; j++) {
            cin >> grid[i][j];

            if (grid[i][j] == 2) {
                sr = i;
                sc = j;
            }
        }
    }

    bfs(sr, sc);

    for (int i = 0; i < n; i++) {
        for (int j = 0; j < m; j++) {
            cout << dist[i][j] << ' ';
        }
        cout << '\n';
    }
}
```
